     8→function resolvePath(cwd: string, target: string): string {
     9→  if (target.startsWith('/')) return normalizePath(target);
    10→  if (target === '..') {
    11→    const parts = cwd.split('/').filter(Boolean);
    12→    parts.pop();
    13→    return '/' + parts.join('/');
    14→  }
    15→  if (target === '.') return cwd;
    16→  return normalizePath(cwd === '/' ? `/${target}` : `${cwd}/${target}`);
    17→}
    18→
    19→function normalizePath(p: string): string {
    20→  const parts = p.split('/').filter(Boolean);
    21→  const resolved: string[] = [];
    22→  for (const part of parts) {
    23→    if (part === '..') resolved.pop();
    24→    else if (part !== '.') resolved.push(part);
    25→  }
    26→  return '/' + resolved.join('/');
    27→}
    28→
    29→function getNode(fs: FSNode, path: string): FSNode | null {
    30→  if (path === '/') return fs;
    31→  const parts = path.split('/').filter(Boolean);
    32→  let current = fs;
    33→  for (const part of parts) {
    34→    if (!current.children || !current.children[part]) return null;
    35→    current = current.children[part];
    36→  }
    37→  return current;
    38→}

    40→export function executeCommand(
    41→  rawInput: string,
    42→  ctx: CommandContext
    43→): OutputLine[] {
    44→  const trimmed = rawInput.trim();
    45→  if (!trimmed) return [];
    46→
    47→  const parts = trimmed.split(/\s+/);
    48→  const cmd = parts[0].toLowerCase();
    49→  const args = parts.slice(1);
    50→
    51→  const commands: Record<string, (args: string[], ctx: CommandContext) => OutputLine[]> = {
    52→    ls: cmdLs,
    53→    cd: cmdCd,
    54→    cat: cmdCat,
    55→    whoami: cmdWhoami,
    56→    skills: cmdSkills,
    57→    experience: cmdExperience,
    58→    education: cmdEducation,
    59→    github: cmdGithub,
    60→    contact: cmdContact,
    61→    resume: cmdResume,
    62→    theme: cmdTheme,
    63→    crt: cmdCrt,
    64→    keys: cmdKeys,
    65→    clear: cmdClear,
    66→    help: cmdHelp,
    67→    projects: cmdProjects,
    68→    profile: cmdProfile,
    69→  };
    70→
    71→  const handler = commands[cmd];
    72→  if (handler) return handler(args, ctx);
    73→
    74→  return [{ id: uid(), content: `retrosh: command not found: ${cmd}. Type 'help' for available commands.`, type: 'error' }];
    75→}

    77→function cmdLs(args: string[], ctx: CommandContext): OutputLine[] {
    78→  const target = args[0] ? resolvePath(ctx.cwd, args[0]) : ctx.cwd;
    79→  const node = getNode(ctx.fs, target);
    80→  if (!node) return [{ id: uid(), content: `ls: cannot access '${args[0] || target}': No such file or directory`, type: 'error' }];
    81→  if (node.type === 'file') return [{ id: uid(), content: node.name, type: 'output' }];
    82→  if (!node.children) return [{ id: uid(), content: '(empty directory)', type: 'output' }];
    83→  const entries = Object.keys(node.children).sort((a, b) => {
    84→    const aDir = node.children![a].type === 'directory' ? 0 : 1;
    85→    const bDir = node.children![b].type === 'directory' ? 0 : 1;
    86→    if (aDir !== bDir) return aDir - bDir;
    87→    return a.localeCompare(b);
    88→  });
    89→  return entries.map(name => {
    90→    const child = node.children![name];
    91→    const suffix = child.type === 'directory' ? '/' : '';
    92→    return { id: uid(), content: `  ${name}${suffix}`, type: 'output' as const };
    93→  });
    94→}

    96→function cmdCd(args: string[], ctx: CommandContext): OutputLine[] {
    97→  const target = args[0] || '/';
    98→  const resolved = resolvePath(ctx.cwd, target);
    99→  const node = getNode(ctx.fs, resolved);
   100→  if (!node) return [{ id: uid(), content: `cd: no such file or directory: ${target}`, type: 'error' }];
   101→  if (node.type === 'file') return [{ id: uid(), content: `cd: not a directory: ${target}`, type: 'error' }];
   102→  ctx.setCwd(resolved || '/');
   103→  return [];
   104→}

   106→function cmdCat(args: string[], ctx: CommandContext): OutputLine[] {
   107→  if (!args[0]) return [{ id: uid(), content: 'cat: missing file operand', type: 'error' }];
   108→  const resolved = resolvePath(ctx.cwd, args[0]);
   109→  const node = getNode(ctx.fs, resolved);

   111→  if (node.type === 'directory') return [{ id: uid(), content: `cat: ${args[0]}: Is a directory`, type: 'error' }];
   112→  return node.content!.split('\n').map(line => ({ id: uid(), content: line, type: 'output' as const }));

   214→  try {

   216→      fetch('https://api.github.com/users/Markes10'),

   582→function cmdTheme(args: string[], ctx: CommandContext): OutputLine[] {
   583→  const validThemes: ThemeName[] = ['amber', 'green', 'white'];
   584→  if (!args[0]) return [{ id: uid(), content: 'theme: usage: theme <amber|green|white>', type: 'output' }];
   585→  const t = args[0].toLowerCase() as ThemeName;
   586→  if (!validThemes.includes(t)) {
   587→    return [{ id: uid(), content: `theme: invalid theme '${args[0]}'. Available: amber, green, white`, type: 'error' }];
   588→  }
   589→  ctx.setTheme(t);
   590→  return [{ id: uid(), content: `theme: switched to ${t}`, type: 'output' }];
   591→}

   647→// ── Tab / ghost-text completion ─────────────────────────────────────────
   648→const ALL_CMDS = ['ls','cd','cat','whoami','skills','experience','education','github','contact','resume','theme','crt','keys','clear','help','projects','profile'];
   649→
   650→export function getTabCompletion(input: string, cwd: string, fs: FSNode): string {
   651→  const parts = input.split(/\s+/);
   652→  if (parts.length <= 1) {
   653→    const m = ALL_CMDS.find(c => c.startsWith(parts[0]));
   654→    return m ? m : input;
   655→  }
   656→  const cmd = parts[0];
   657→  if (!['ls','cd','cat'].includes(cmd)) return input;
   658→  const pathPartial = parts[parts.length - 1];
   659→  const dirParts = pathPartial.split('/');
   660→  const prefix = dirParts.pop() || '';
   661→  const dirPath = dirParts.join('/');
   662→  const targetDir = dirPath ? resolvePath(cwd, dirPath) : cwd;
   663→  const node = getNode(fs, targetDir);
   664→  if (!node || node.type !== 'directory' || !node.children) return input;
   665→  const matches = Object.keys(node.children).filter(n => n.startsWith(prefix));
   666→  if (matches.length === 1) {
   667→    const c = matches[0];
   668→    const sfx = node.children[c].type === 'directory' ? '/' : '';
   669→    return parts.slice(0, -1).join(' ') + ' ' + dirPath + (dirPath ? '/' : '') + c + sfx;
   670→  }
   671→  return input;
   672→}
   673→
   674→export function getGhostText(input: string, cwd: string, fs: FSNode): string {
   675→  const parts = input.split(/\s+/);
   676→  if (parts.length <= 1) {
   677→    const m = ALL_CMDS.find(c => c.startsWith(parts[0]));
   678→    return m ? m.slice(parts[0].length) : '';
   679→  }
   680→  const cmd = parts[0];
   681→  if (!['ls','cd','cat'].includes(cmd)) return '';
   682→  const pathPartial = parts[parts.length - 1];
   683→  const dirParts = pathPartial.split('/');
   684→  const prefix = dirParts.pop() || '';
   685→  const dirPath = dirParts.join('/');
   686→  const targetDir = dirPath ? resolvePath(cwd, dirPath) : cwd;
   687→  const node = getNode(fs, targetDir);
   688→  if (!node || node.type !== 'directory' || !node.children) return '';
   689→  const matches = Object.keys(node.children).filter(n => n.startsWith(prefix));
   690→  if (matches.length === 1) {
   691→    const c = matches[0];
   692→    return c.slice(prefix.length) + (node.children[c].type === 'directory' ? '/' : '');
   693→  }
   694→  return '';
   695→}
