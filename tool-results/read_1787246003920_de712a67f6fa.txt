     8→     8→function resolvePath(cwd: string, target: string): string {
     9→     9→  if (target.startsWith('/')) return normalizePath(target);
    10→    10→  if (target === '..') {
    11→    11→    const parts = cwd.split('/').filter(Boolean);
    12→    12→    parts.pop();
    13→    13→    return '/' + parts.join('/');
    14→    14→  }
    15→    15→  if (target === '.') return cwd;
    16→    16→  return normalizePath(cwd === '/' ? `/${target}` : `${cwd}/${target}`);
    17→    17→}
    18→    18→
    19→    19→function normalizePath(p: string): string {
    20→    20→  const parts = p.split('/').filter(Boolean);
    21→    21→  const resolved: string[] = [];
    22→    22→  for (const part of parts) {
    23→    23→    if (part === '..') resolved.pop();
    24→    24→    else if (part !== '.') resolved.push(part);
    25→    25→  }
    26→    26→  return '/' + resolved.join('/');
    27→    27→}
    28→    28→
    29→    29→function getNode(fs: FSNode, path: string): FSNode | null {
    30→    30→  if (path === '/') return fs;
    31→    31→  const parts = path.split('/').filter(Boolean);
    32→    32→  let current = fs;
    33→    33→  for (const part of parts) {
    34→    34→    if (!current.children || !current.children[part]) return null;
    35→    35→    current = current.children[part];
    36→    36→  }
    37→    37→  return current;
    38→    38→}

    77→    77→function cmdLs(args: string[], ctx: CommandContext): OutputLine[] {
    78→    78→  const target = args[0] ? resolvePath(ctx.cwd, args[0]) : ctx.cwd;
    79→    79→  const node = getNode(ctx.fs, target);
    80→    80→  if (!node) return [{ id: uid(), content: `ls: cannot access '${args[0] || target}': No such file or directory`, type: 'error' }];
    81→    81→  if (node.type === 'file') return [{ id: uid(), content: node.name, type: 'output' }];
    82→    82→  if (!node.children) return [{ id: uid(), content: '(empty directory)', type: 'output' }];
    83→    83→  const entries = Object.keys(node.children).sort((a, b) => {
    84→    84→    const aDir = node.children![a].type === 'directory' ? 0 : 1;
    85→    85→    const bDir = node.children![b].type === 'directory' ? 0 : 1;
    86→    86→    if (aDir !== bDir) return aDir - bDir;
    87→    87→    return a.localeCompare(b);
    88→    88→  });
    89→    89→  return entries.map(name => {
    90→    90→    const child = node.children![name];
    91→    91→    const suffix = child.type === 'directory' ? '/' : '';
    92→    92→    return { id: uid(), content: `  ${name}${suffix}`, type: 'output' as const };
    93→    93→  });
    94→    94→}

    96→    96→function cmdCd(args: string[], ctx: CommandContext): OutputLine[] {

   101→   101→  if (node.type === 'file') return [{ id: uid(), content: `cd: not a directory: ${target}`, type: 'error' }];
   102→   102→  ctx.setCwd(resolved || '/');
   103→   103→  return [];
   104→   104→}

   648→   648→const ALL_CMDS = ['ls','cd','cat','whoami','skills','experience','education','github','contact','resume','theme','crt','keys','clear','help','projects','profile'];
   649→   649→
   650→   650→export function getTabCompletion(input: string, cwd: string, fs: FSNode): string {
   651→   651→  const parts = input.split(/\s+/);
   652→   652→  if (parts.length <= 1) {
   653→   653→    const m = ALL_CMDS.find(c => c.startsWith(parts[0]));
   654→   654→    return m ? m : input;
   655→   655→  }
   656→   656→  const cmd = parts[0];
   657→   657→  if (!['ls','cd','cat'].includes(cmd)) return input;
   658→   658→  const pathPartial = parts[parts.length - 1];
   659→   659→  const dirParts = pathPartial.split('/');
   660→   660→  const prefix = dirParts.pop() || '';
   661→   661→  const dirPath = dirParts.join('/');
   662→   662→  const targetDir = dirPath ? resolvePath(cwd, dirPath) : cwd;
   663→   663→  const node = getNode(fs, targetDir);
   664→   664→  if (!node || node.type !== 'directory' || !node.children) return input;
   665→   665→  const matches = Object.keys(node.children).filter(n => n.startsWith(prefix));
   666→   666→  if (matches.length === 1) {
   667→   667→    const c = matches[0];
   668→   668→    const sfx = node.children[c].type === 'directory' ? '/' : '';
   669→   669→    return parts.slice(0, -1).join(' ') + ' ' + dirPath + (dirPath ? '/' : '') + c + sfx;
   670→   670→  }
   671→   671→  return input;
   672→   672→}

   674→   674→export function getGhostText(input: string, cwd: string, fs: FSNode): string {
   675→   675→  const parts = input.split(/\s+/);
   676→   676→  if (parts.length <= 1) {
   677→   677→    const m = ALL_CMDS.find(c => c.startsWith(parts[0]));
   678→   678→    return m ? m.slice(parts[0].length) : '';
   679→   679→  }
   680→   680→  const cmd = parts[0];
   681→   681→  if (!['ls','cd','cat'].includes(cmd)) return '';
   682→   682→  const pathPartial = parts[parts.length - 1];
   683→   683→  const dirParts = pathPartial.split('/');
   684→   684→  const prefix = dirParts.pop() || '';
   685→   685→  const dirPath = dirParts.join('/');
   686→   686→  const targetDir = dirPath ? resolvePath(cwd, dirPath) : cwd;
   687→   687→  const node = getNode(fs, targetDir);
   688→   688→  if (!node || node.type !== 'directory' || !node.children) return '';
   689→   689→  const matches = Object.keys(node.children).filter(n => n.startsWith(prefix));
   690→   690→  if (matches.length === 1) {
   691→   691→    const c = matches[0];
   692→   692→    return c.slice(prefix.length) + (node.children[c].type === 'directory' ? '/' : '');
   693→   693→  }
   694→   694→  return '';
   695→   695→}
